static VALUE boltffi_ruby_decode_string(FfiBuf_u8 buffer) {
    if (buffer.ptr == NULL || buffer.len < sizeof(uint32_t)) {
        boltffi_free_buf(buffer);
        rb_raise(rb_eRuntimeError, "native function returned truncated string buffer");
    }

    uint32_t wire_len = boltffi_ruby_read_u32_le(buffer.ptr);
    if ((uintptr_t)wire_len > buffer.len - sizeof(uint32_t)) {
        boltffi_free_buf(buffer);
        rb_raise(rb_eRuntimeError, "native function returned truncated string buffer");
    }

    boltffi_ruby_string_build state = { (const char *)buffer.ptr + sizeof(uint32_t), (long)wire_len };
    int error = 0;
    VALUE value = rb_protect(boltffi_ruby_build_utf8_string_protected, (VALUE)(uintptr_t)&state, &error);
    boltffi_free_buf(buffer);
    if (error) rb_jump_tag(error);
    return value;
}

typedef struct {
    const uint8_t *ptr;
    const uint8_t *end;
} boltffi_ruby_wire_reader;

static boltffi_ruby_wire_reader boltffi_ruby_wire_reader_new(FfiBuf_u8 buffer) {
    boltffi_ruby_wire_reader reader;
    reader.ptr = buffer.ptr;
    reader.end = buffer.ptr == NULL ? NULL : buffer.ptr + buffer.len;
    return reader;
}

static int boltffi_ruby_wire_read_exact(boltffi_ruby_wire_reader *reader, uintptr_t len, const uint8_t **out) {
    if (reader->ptr == NULL || reader->end == NULL || (uintptr_t)(reader->end - reader->ptr) < len) return 0;
    *out = reader->ptr;
    reader->ptr += len;
    return 1;
}

static int boltffi_ruby_wire_read_u8(boltffi_ruby_wire_reader *reader, uint8_t *out) {
    const uint8_t *bytes;
    if (!boltffi_ruby_wire_read_exact(reader, 1, &bytes)) return 0;
    *out = bytes[0];
    return 1;
}

static int boltffi_ruby_wire_read_i8(boltffi_ruby_wire_reader *reader, int8_t *out) {
    uint8_t value;
    if (!boltffi_ruby_wire_read_u8(reader, &value)) return 0;
    *out = (int8_t)value;
    return 1;
}

static int boltffi_ruby_wire_read_bool(boltffi_ruby_wire_reader *reader, int *out) {
    uint8_t value;
    if (!boltffi_ruby_wire_read_u8(reader, &value)) return 0;
    if (value > 1) return 0;
    *out = value != 0;
    return 1;
}

static int boltffi_ruby_wire_read_u16(boltffi_ruby_wire_reader *reader, uint16_t *out) {
    const uint8_t *bytes;
    if (!boltffi_ruby_wire_read_exact(reader, sizeof(uint16_t), &bytes)) return 0;
    *out = boltffi_ruby_read_u16_le(bytes);
    return 1;
}

static int boltffi_ruby_wire_read_i16(boltffi_ruby_wire_reader *reader, int16_t *out) {
    uint16_t value;

    return 1;
}

static int boltffi_ruby_wire_read_f32(boltffi_ruby_wire_reader *reader, float *out) {
    uint32_t bits;
    if (!boltffi_ruby_wire_read_u32(reader, &bits)) return 0;
    memcpy(out, &bits, sizeof(bits));
    return 1;
}

static int boltffi_ruby_wire_read_f64(boltffi_ruby_wire_reader *reader, double *out) {
    uint64_t bits;

    return 1;
}

static int boltffi_ruby_wire_read_bytes(boltffi_ruby_wire_reader *reader, uint8_t **out_ptr, uintptr_t *out_len) {
    uint32_t len;
    const uint8_t *bytes;
    if (!boltffi_ruby_wire_read_u32(reader, &len)) return 0;
    if (!boltffi_ruby_wire_read_exact(reader, len, &bytes)) return 0;
    uint8_t *copy = NULL;
    if (len > 0) {
        copy = (uint8_t *)malloc((size_t)len);
        if (copy == NULL) return 0;
        memcpy(copy, bytes, len);
    }
    *out_ptr = copy;
    *out_len = (uintptr_t)len;
    return 1;
}

static int boltffi_ruby_wire_read_string(boltffi_ruby_wire_reader *reader, char **out_ptr, uintptr_t *out_len) {
    uint8_t *bytes = NULL;
    if (!boltffi_ruby_wire_read_bytes(reader, &bytes, out_len)) return 0;
    *out_ptr = (char *)bytes;
    return 1;
}
